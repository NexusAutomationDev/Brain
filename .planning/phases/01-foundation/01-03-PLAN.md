```typescript
import pino from 'pino';

export interface LogContext {
  tenantId?: string;
  brainId?: string;
  sessionId?: string;
  userId?: string;
}

export function createLogger(context: LogContext = {}) {
  return pino({
    level: process.env.LOG_LEVEL || 'info',
    formatters: {
      level: (label) => ({ level: label }),
    },
    timestamp: pino.stdTimeFunctions.isoTime,
    base: {
      // OBS-01: Include context in all log lines
      ...context,
      env: process.env.NODE_ENV || 'development',
    },
  });
}
```

This implements OBS-01 requirement: structured JSON logging with timestamps, level, and context (tenant, Brain, session, user).

```typescript
import type { Sql } from 'postgres';

export interface HealthCheckResult {
  status: 'ok' | 'degraded' | 'error';
  checks: {
    db: 'connected' | 'failed';
  };
  version?: string;      // D-15 (Claude's discretion): include version info
  timestamp: string;
}

export async function checkDatabase(sql: Sql): Promise<boolean> {
  try {
    await sql`SELECT 1`;
    return true;
  } catch {
    return false;
  }
}

export async function performHealthCheck(sql: Sql): Promise<HealthCheckResult> {
  const dbOk = await checkDatabase(sql);

  // D-13: Structured JSON response with status and checks
  return {
    status: dbOk ? 'ok' : 'error',
    checks: {
      db: dbOk ? 'connected' : 'failed',
    },
    version: process.env.GIT_COMMIT || 'unknown',
    timestamp: new Date().toISOString(),
  };
}

/**
 * HTTP status code mapping (D-14):
 * - 200: status === 'ok'
 * - 503: checks.db === 'failed'
 * - 500: status === 'error' (internal error)
 *
 * Note: D-15 defers transport: "webhook" to Phase 2 when transport package exists
 */
```

This implements OBS-02 requirement: health check endpoint returning database status with granular HTTP status codes per D-14.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Application → Logs | Log output may contain sensitive data; must redact credentials |
| Public → Health endpoint | Health check exposes system state; must not leak credentials or internal details |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Information Disclosure | Logs containing credentials | mitigate | Never pass DATABASE_URL or secrets to logger context; pino formatters redact by default; future: add redact config for sensitive fields |
| T-03-02 | Information Disclosure | Health check exposing DB credentials | mitigate | Health check returns only status and boolean checks, no connection strings or usernames; version string is commit hash only |
| T-03-03 | Denial of Service | Health check query timeout | accept | SELECT 1 query is fast (<1ms); postgres.js has built-in query timeouts; low risk |
| T-03-04 | Information Disclosure | Tenant context leakage in logs | accept | Logger explicitly injects tenantId/brainId — this is intentional for debugging; logs are internal-only (not exposed to end users) |

</threat_model>
